# Face matching via cosine similarity (find_best_match) def find_best_match(face_embedding, known, names, ids, threshold=0.65): face_embedding = face_embedding.astype(np.float32) # normalize vectors norm = np.linalg.norm(face_embedding) face_embedding /= norm known_array = np.array(known, dtype=np.float32) known_array /= np.linalg.norm(known_array, axis=1, keepdims=True) # cosine similarity sims = np.dot(known_array, face_embedding) best_idx = int(np.argmax(sims)) if sims[best_idx] >= threshold: return names[best_idx], ids[best_idx], sims[best_idx]
